/* ---------- helpers ---------- */
const $ = (s, el = document) => el.querySelector(s);
const esc = s => String(s ?? "").replace(/[&<>"]/g, c => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;" }[c]));
const nf = (v, d = 2) => v == null ? "n/a" : Number(v).toLocaleString("en-US", { minimumFractionDigits: d, maximumFractionDigits: d });
const sg = (v, d = 2) => v == null ? "n/a" : (v > 0 ? "+" : v < 0 ? "−" : "") + nf(Math.abs(v), d);
const cls = v => v == null || v === 0 ? "flat" : v > 0 ? "up" : "down";
const arrow = v => v == null || v === 0 ? "" : v > 0 ? "▲ " : "▼ ";
const cv = n => `var(--${n})`;
const css = n => getComputedStyle(document.documentElement).getPropertyValue("--" + n).trim();

const ICON = {

  watch:   `<svg class="ic" viewBox="0 0 16 16" aria-hidden="true"><path d="M8 2 15 14H1z" fill="var(--warn)"/></svg>`,
  neutral: `<svg class="ic" viewBox="0 0 16 16" aria-hidden="true"><rect x="3" y="7" width="10" height="2.4" rx="1.2" fill="var(--axis)"/></svg>`,
  alert:   `<svg class="ic" viewBox="0 0 16 16" aria-hidden="true"><path d="M8 1 15 8 8 15 1 8z" fill="var(--serious)"/></svg>`,
  stress:  `<svg class="ic" viewBox="0 0 16 16" aria-hidden="true"><path d="M8 1 15 8 8 15 1 8z" fill="var(--serious)"/></svg>`,
  crisis:  `<svg class="ic" viewBox="0 0 16 16" aria-hidden="true"><rect x="2" y="2" width="12" height="12" rx="2" fill="var(--crit)"/><path d="M5.5 5.5l5 5m0-5l-5 5" stroke="#fff" stroke-width="1.8" stroke-linecap="round"/></svg>`,
  na:      `<svg class="ic" viewBox="0 0 16 16" aria-hidden="true"><circle cx="8" cy="8" r="5" fill="none" stroke="var(--axis)" stroke-width="1.5"/></svg>`,

const STATE_LABEL = { ok: "OK", watch: "Watch", alert: "Alert", crisis: "Crisis", na: "No data", support: "Supports", neutral: "Neutral", stress: "Stress" };

function nice(min, max, n = 4) {
  const span = (max - min) || 1, raw = span / n, p = Math.pow(10, Math.floor(Math.log10(raw)));
  const step = [1, 2, 2.5, 5, 10].map(m => m * p).find(s => s >= raw);
  const lo = Math.floor(min / step) * step, hi = Math.ceil(max / step) * step, t = [];
  for (let v = lo; v <= hi + step / 2; v += step) t.push(+v.toFixed(10));
  return { lo, hi, ticks: t };
}

function spark(values, { w = 90, h = 26, color = "var(--s1)" } = {}) {
  const v = (values || []).filter(x => x != null);
  if (v.length < 2) return "";
  const min = Math.min(...v), max = Math.max(...v), r = (max - min) || 1;
  const pts = v.map((y, i) => [2 + i * (w - 4) / (v.length - 1), 2 + (h - 4) * (1 - (y - min) / r)]);
  const d = pts.map((p, i) => (i ? "L" : "M") + p[0].toFixed(1) + " " + p[1].toFixed(1)).join("");
  const last = pts[pts.length - 1];
  return `<svg width="${w}" height="${h}" viewBox="0 0 ${w} ${h}" role="img" aria-label="Trend, from ${nf(v[0])} to ${nf(v[v.length - 1])}"><path d="${d}" fill="none" stroke="${color}" stroke-width="2" stroke-linejoin="round" stroke-linecap="round"/><circle cx="${last[0]}" cy="${last[1]}" r="3" fill="${color}" stroke="var(--surf)" stroke-width="1.5"/></svg>`;
}

/* Responsive line chart with crosshair and tooltip. cfg:
   cats: x labels (array)    series: [{name, values, color, dash, label}]
   yfmt: fn   height   ref: [{v, label}]   bands: [{from, to}] (shaded)   markers: bool */
function lineChart(el, cfg) {
  const draw = () => {
    const W = Math.max(280, el.clientWidth), H = cfg.height || 240;
    const m = { l: 46, r: cfg.endLabels ? 74 : 16, t: 12, b: 28 };
    const all = cfg.series.flatMap(s => s.values.filter(v => v != null)).concat((cfg.ref || []).map(r => r.v));
    const sc = nice(Math.min(...all), Math.max(...all));
    const n = cfg.cats.length, iw = W - m.l - m.r, ih = H - m.t - m.b;
    const X = i => m.l + (n === 1 ? iw / 2 : i * iw / (n - 1)), Y = v => m.t + ih * (1 - (v - sc.lo) / (sc.hi - sc.lo));
    let g = "";
    sc.ticks.forEach(t => { g += `<line x1="${m.l}" x2="${W - m.r}" y1="${Y(t)}" y2="${Y(t)}" stroke="var(--grid)"/><text x="${m.l - 8}" y="${Y(t) + 4}" text-anchor="end">${(cfg.yfmt || (x => nf(x, 0)))(t)}</text>`; });
    (cfg.bands || []).forEach(b => { g += `<rect x="${m.l}" y="${Y(Math.min(b.to, sc.hi))}" width="${iw}" height="${Math.max(0, Y(Math.max(b.from, sc.lo)) - Y(Math.min(b.to, sc.hi)))}" fill="var(--wash)"/>`; });
    (cfg.ref || []).forEach(r => { g += `<line x1="${m.l}" x2="${W - m.r}" y1="${Y(r.v)}" y2="${Y(r.v)}" stroke="var(--axis)" stroke-dasharray="4 4"/><text class="halo" x="${W - m.r - 4}" y="${Y(r.v) - 5}" text-anchor="end">${esc(r.label)}</text>`; });
    const step = Math.max(1, Math.ceil(n / (W < 480 ? 4 : 7)));
    cfg.cats.forEach((c, i) => { if (i % step === 0 || i === n - 1) g += `<text x="${X(i)}" y="${H - 8}" text-anchor="${i === 0 ? "start" : i === n - 1 ? "end" : "middle"}">${esc(cfg.xlab ? cfg.xlab(c) : c)}</text>`; });
    g += `<line x1="${m.l}" x2="${W - m.r}" y1="${Y(sc.lo)}" y2="${Y(sc.lo)}" stroke="var(--axis)"/>`;
    cfg.series.forEach(s => {
      const pts = s.values.map((v, i) => v == null ? null : [X(i), Y(v)]).filter(Boolean);
      if (!pts.length) return;
      g += `<path d="${pts.map((p, i) => (i ? "L" : "M") + p[0].toFixed(1) + " " + p[1].toFixed(1)).join("")}" fill="none" stroke="${s.color}" stroke-width="${s.width || 2}" ${s.dash ? `stroke-dasharray="${s.dash}"` : ""} stroke-linejoin="round" stroke-linecap="round"/>`;
      if (cfg.markers) pts.forEach(p => { g += `<circle cx="${p[0]}" cy="${p[1]}" r="4.5" fill="${s.color}" stroke="var(--surf)" stroke-width="2"/>`; });
      if (cfg.endLabels && s.label) { const l = pts[pts.length - 1]; g += `<text class="halo ink" x="${l[0] + 8}" y="${l[1] + 4}">${esc(s.label)}</text>`; }
    });
    g += `<line id="xh" y1="${m.t}" y2="${m.t + ih}" stroke="var(--ink2)" stroke-dasharray="3 3" style="display:none"/>`;
    cfg.series.forEach((s, k) => { g += `<circle class="hd" data-k="${k}" r="5" fill="${s.color}" stroke="var(--surf)" stroke-width="2" style="display:none"/>`; });
    g += `<rect id="hit" x="${m.l}" y="${m.t}" width="${iw}" height="${ih}" fill="transparent"/>`;
    el.innerHTML = `<svg viewBox="0 0 ${W} ${H}" width="${W}" height="${H}" role="img" aria-label="${esc(cfg.aria || "Chart")}">${g}</svg><div class="tip"></div>`;
    const svg = $("svg", el), tip = $(".tip", el), xh = $("#xh", el), hit = $("#hit", el), hds = [...el.querySelectorAll(".hd")];
    const show = ev => {
      const rc = svg.getBoundingClientRect(), px = (ev.clientX - rc.left) * (W / rc.width);
      const i = Math.max(0, Math.min(n - 1, Math.round((px - m.l) / (iw / Math.max(1, n - 1)))));
      xh.setAttribute("x1", X(i)); xh.setAttribute("x2", X(i)); xh.style.display = "";
      hds.forEach((c, k) => { const v = cfg.series[k].values[i]; if (v == null) { c.style.display = "none"; return; } c.setAttribute("cx", X(i)); c.setAttribute("cy", Y(v)); c.style.display = ""; });
      tip.innerHTML = `<b>${esc(cfg.xlab ? cfg.xlab(cfg.cats[i]) : cfg.cats[i])}</b>` + cfg.series.map(s => `<div class="tl"><span class="sw" style="border-color:${s.color}"></span>${cfg.series.length > 1 ? esc(s.name) + ": " : ""}<b>${s.values[i] == null ? "n/a" : (cfg.tipfmt || cfg.yfmt || nf)(s.values[i])}</b></div>`).join("");
      tip.style.display = "block";
      const tw = tip.offsetWidth, x = X(i) * (rc.width / W);
      tip.style.left = Math.max(0, Math.min(rc.width - tw, x - tw / 2)) + "px"; tip.style.top = "0px";
    };
    const hide = () => { tip.style.display = "none"; xh.style.display = "none"; hds.forEach(c => c.style.display = "none"); };
    hit.addEventListener("pointermove", show); hit.addEventListener("pointerdown", show); hit.addEventListener("pointerleave", hide);
  };
  draw();
  let lastW = el.clientWidth;
  new ResizeObserver(() => { if (Math.abs(el.clientWidth - lastW) > 2) { lastW = el.clientWidth; draw(); } }).observe(el);
}

function renderRegime(d) {
  const g = d.regime;
  if (!g) return `<div class="card"><h3>Macro regime</h3><p class="muted small">Not enough data to score the regime.</p></div>`;
  const keys = Object.keys(g.dims), n = keys.length, cx = 240, cy = 130, R = 86;
  const pt = (i, v) => { const a = -Math.PI / 2 + i * 2 * Math.PI / n, r = R * v / 100; return [cx + r * Math.cos(a), cy + r * Math.sin(a)]; };
  const poly = f => keys.map((k, i) => { const v = f(g.dims[k]); return v == null ? null : pt(i, v).map(x => x.toFixed(1)).join(","); }).filter(Boolean).join(" ");

  if (g.avg_prev != null) s += `<polygon points="${poly(x => x.prev)}" fill="none" stroke="var(--ink2)" stroke-width="1.5" stroke-dasharray="5 4"/>`;
  s += `<polygon points="${poly(x => x.now)}" fill="var(--s1)" fill-opacity=".22" stroke="var(--s1)" stroke-width="2" stroke-linejoin="round"/>`;
  keys.forEach((k, i) => {
    const p = pt(i, g.dims[k].now), a = -Math.PI / 2 + i * 2 * Math.PI / n, lx = cx + (R + 16) * Math.cos(a), ly = cy + (R + 16) * Math.sin(a);
    s += `<circle cx="${p[0]}" cy="${p[1]}" r="4.5" fill="var(--s1)" stroke="var(--surf)" stroke-width="2"/>`;
    const anc = Math.abs(Math.cos(a)) < .25 ? "middle" : Math.cos(a) > 0 ? "start" : "end";
    s += `<text class="ink" x="${lx}" y="${ly + (Math.sin(a) > .5 ? 12 : Math.sin(a) < -.5 ? -4 : 4)}" text-anchor="${anc}">${esc(g.dims[k].name)} <tspan font-weight="650">${g.dims[k].now}</tspan></text>`;
  });
  const dirTxt = { improving: "Improving", deteriorating: "Deteriorating", stable: "Stable" }[g.direction];
  const lab = { "risk-on": "Risk-on", "risk-off": "Risk-off", mixed: "Mixed" }[g.label];
  return `<div class="card">
    <div class="cardhead"><h3>Macro regime</h3><span class="muted small">Average ${g.avg}${g.avg_prev != null ? `, was ${g.avg_prev} a month ago` : ""}</span></div>
    <div class="rrow"><span class="rlabel">${lab}</span><span class="chip">${dirTxt}</span></div>
    <p class="muted small">Weakest: ${esc(g.weakest.join(" and "))}. ${g.spread >= 40 ? "The dimensions disagree, so read the shape, not the average." : ""}</p>
    <svg class="radar" viewBox="0 0 480 266" role="img" aria-label="Regime radar. ${keys.map(k => g.dims[k].name + " " + g.dims[k].now).join(", ")}">${s}</svg>
    <div class="legend"><span class="lg"><span class="sw" style="border-color:var(--s1)"></span>Now</span>${g.avg_prev != null ? `<span class="lg"><span class="sw" style="border-color:var(--ink2);border-top-style:dashed"></span>About a month ago</span>` : ""}</div>
    <details class="tbl"><summary>Show scores</summary><div class="scroll"><table><tr><th>Dimension</th><th>Now</th><th>Month ago</th><th>Based on</th></tr>${keys.map(k => `<tr><td>${esc(g.dims[k].name)}</td><td>${g.dims[k].now}</td><td>${g.dims[k].prev ?? "n/a"}</td><td style="text-align:left">${esc(g.dims[k].detail)}</td></tr>`).join("")}</table></div></details>
  </div>`;
}

  if (d.credit && $("#ch-hy")) {

    lineChart($("#ch-hy"), { cats: c.dates, height: 200, aria: "High-yield spread over the past year", yfmt: x => nf(x, 0), tipfmt: x => nf(x, 0) + " bp", xlab: s => { const t = new Date(s + "T00:00:00Z"); return t.toLocaleDateString("en-US", { month: "short", year: "2-digit", timeZone: "UTC" }); },
      ref: [{ v: 450, label: "450 concern" }, { v: 300, label: "300 euphoria" }], series: [{ name: "HY spread", values: c.spark.map(x => x * 100), color: css("s1"), width: 2 }] });
  }
  if (d.vol && $("#ch-vix")) {
    const v = d.vol;
    lineChart($("#ch-vix"), { cats: v.curve.map(x => x.t), height: 190, aria: "VIX term structure", yfmt: x => nf(x, 0), tipfmt: x => nf(x, 1), markers: true, series: [{ name: "Implied volatility", values: v.curve.map(x => x.v), color: css("s1"), width: 2 }] });
  }
}

function initTheme() {
  const root = document.documentElement, btn = $("#theme");
  let saved = null; try { saved = localStorage.getItem("mc-theme"); } catch (e) { /* ignore */ }
  if (saved) root.setAttribute("data-theme", saved);
  const label = () => { const t = root.getAttribute("data-theme"); btn.textContent = t === "dark" ? "Theme: dark" : t === "light" ? "Theme: light" : "Theme: auto"; };
  label();
  btn.addEventListener("click", () => {
    const t = root.getAttribute("data-theme"), nx = t === null ? "dark" : t === "dark" ? "light" : null;
    if (nx) root.setAttribute("data-theme", nx); else root.removeAttribute("data-theme");
    try { nx ? localStorage.setItem("mc-theme", nx) : localStorage.removeItem("mc-theme"); } catch (e) { /* ignore */ }
    label(); render(window.__d);
  });
}
